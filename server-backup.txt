app.post("/api/admin/login", (req, res) => {
    const { username, password } = req.body || {};

    if (
        username === process.env.ADMIN_USERNAME &&
        password === process.env.ADMIN_PASSWORD
    ) {
        return res.json({
            success: true,

        });

// Generate QR
app.post("/api/qr/generate", async (req, res) => {
    try {
        const studentId =
            typeof req.body?.studentId === "string"
                ? req.body.studentId.trim()
                : "";

        const eventId = Number(req.body?.eventId);

        if (
            !studentId ||
            !Number.isSafeInteger(eventId) ||
            eventId <= 0
        ) {
            return res.status(400).json({
                success: false,
                error: "Enter a valid Student ID and Event ID"
            });
        }

        const student = attendanceDb.prepare(`
            SELECT id, student_id, name, email
            FROM students
            WHERE student_id = ?
        `).get(studentId);

        if (!student) {
            return res.status(404).json({
                success: false,
                error: "Student ID was not found"
            });
        }

        const event = attendanceDb.prepare(`
            SELECT id, name
            FROM events
            WHERE id = ?
        `).get(eventId);

        if (!event) {
            return res.status(404).json({
                success: false,
                error: "Event ID was not found"
            });
        }

        const token = crypto.randomBytes(32).toString("hex");
        const expiresAt = Date.now() + 60 * 1000;

        const qrData = JSON.stringify({
            token,
            expiresAt
        });

        const qrCode = await QRCode.toDataURL(qrData);

        currentQR = {
            token,
            expiresAt,
            studentId: student.id,
            eventId: event.id
        };

        // Return QR even if email delivery fails
        let emailSent = false;
        let emailMessage = "QR generated; email not sent.";

        if (student.email && process.env.EMAIL_USER &&
            process.env.EMAIL_APP_PASSWORD) {
            try {
                await transporter.sendMail({
                    from: process.env.EMAIL_USER,
                    to: student.email,
                    subject: "Smart QR Attendance - Your QR Code",
                    text:
                        `Hello ${student.name || "Student"},\n\n` +
                        "Your Smart QR Attendance QR code has been generated.\n\n" +
                        "This QR code is valid for 60 seconds only.\n\n" +
                        "Please use the original QR code for attendance.",
                    attachments: [
                        {
                            filename: "attendance-qr.png",
                            content: Buffer.from(
                                qrCode.split(",")[1],
                                "base64"
                            ),
                            contentType: "image/png"
                        }
                    ]
                });

                emailSent = true;
                emailMessage = "Email sent successfully.";
            } catch (emailError) {
                console.error(
                    "Email sending failed:",
                    emailError.message
                );
                emailMessage = "QR generated, but email delivery failed.";
            }
        } else {
            emailMessage = "QR generated, but student email or email settings are missing.";
        }

        return res.json({
            success: true,
            qrCode,
            expiresAt,
            emailSent,
            emailMessage
        });

    } catch (error) {
        console.error("QR generation error:", error);

        return res.status(500).json({
            success: false,
            error: "QR generation failed. Check the server terminal."
        });
    }
});
